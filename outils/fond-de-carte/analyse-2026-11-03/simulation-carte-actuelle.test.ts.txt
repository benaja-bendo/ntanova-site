// SIMULATION JETABLE (hors dépôt, dans le clone de lecture) : que devient la carte actuelle (SVG, une page par « coin »)
// quand des dizaines, puis des centaines d'écoles sont vues sur place ? On exécute le vrai code de src/modele/carte/.
import { describe, it } from 'vitest';
import { dansContour, distanceMetres } from '../../src/modele/geo';
import {
  arrondissementDessine,
  cadreArrondissement,
  cadreCoin,
  coinsDe,
  planer,
  TAILLE_MARQUE,
  type EcoleSurCarte,
} from '../../src/modele/carte/plan';
import { boiteDeContours } from '../../src/modele/carte/geometrie';

function graine(a: number) {
  return () => {
    a |= 0;
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function gauss(r: () => number) {
  return Math.sqrt(-2 * Math.log(1 - r())) * Math.cos(2 * Math.PI * r());
}

function ecoles(slug: string, n: number, mode: 'uniforme' | 'centre'): EcoleSurCarte[] {
  const dessin = arrondissementDessine(slug)!;
  const b = boiteDeContours([dessin.contour]);
  const r = graine(42 + n);
  const out: EcoleSurCarte[] = [];
  // Un centre de densité : le milieu du rectangle, décalé vers le sud-ouest (là où se trouvent les rues).
  const cx = (b.ouest + b.est) / 2;
  const cy = (b.sud + b.nord) / 2;
  let essais = 0;
  while (out.length < n && essais++ < 200000) {
    const p =
      mode === 'uniforme'
        ? { longitude: b.ouest + r() * (b.est - b.ouest), latitude: b.sud + r() * (b.nord - b.sud) }
        : { longitude: cx + (gauss(r) * 1500) / 111320, latitude: cy + (gauss(r) * 1500) / 111320 };
    if (!dansContour(p, dessin.contour)) continue;
    out.push({ id: `S${out.length}`, nom: `École ${out.length}`, lien: `/x/${out.length}/`, etat: 'relevee', ...p });
  }
  return out;
}

describe('simulation carte actuelle', () => {
  it('tourne', async () => {
    const lignes: string[] = [];
    for (const slug of ['tie-tie', 'ngoyo', 'lumumba']) {
      const dessin = arrondissementDessine(slug)!;
      const { boite, projection } = cadreArrondissement(dessin);
      const metresMarque = TAILLE_MARQUE.largeur * projection.metresParUnite;
      lignes.push(
        `\n## ${slug} : ${Math.round(projection.metresParUnite)} m par unité du dessin ; deux écoles à moins de ${Math.round(metresMarque)} m (en largeur ET en hauteur) fusionnent`,
      );
      for (const mode of ['uniforme', 'centre'] as const) {
        for (const n of [5, 10, 30, 60, 100, 200, 300, 600, 1000]) {
          const liste = ecoles(slug, n, mode);
          const t0 = performance.now();
          const plan = planer(boite, liste, projection);
          const ms = performance.now() - t0;
          const seules = plan.marques.filter((m) => m.genre === 'ecole').length;
          const groupes = plan.marques.filter((m) => m.genre === 'groupe');
          const dansGroupes = groupes.reduce((s, g) => s + (g.genre === 'groupe' ? g.ecoles.length : 0), 0);
          const coins = coinsDe(dessin, liste);
          // Sur la page d'un coin : combien d'écoles de ce groupe restent dans un groupe (jamais vues seules sur la carte) ?
          let invisibles = 0;
          let plusGrosGroupe = 0;
          let plusLarge = 0;
          for (const g of coins) {
            plusGrosGroupe = Math.max(plusGrosGroupe, g.ecoles.length);
            const c = cadreCoin(g.ecoles);
            const planCoin = planer(c.boite, liste, c.projection);
            const ids = new Set(g.ecoles.map((e) => e.id));
            const seulesDuGroupe = planCoin.marques.filter((m) => m.genre === 'ecole' && ids.has(m.ecole.id)).length;
            invisibles += g.ecoles.length - seulesDuGroupe;
            // étendue du groupe en mètres
            let dmax = 0;
            for (const a of g.ecoles) for (const b2 of g.ecoles) dmax = Math.max(dmax, distanceMetres(a, b2));
            plusLarge = Math.max(plusLarge, dmax);
          }
          lignes.push(
            `${mode.padEnd(8)} N=${String(n).padStart(4)} | maisons seules ${String(seules).padStart(3)} | pastilles ${String(groupes.length).padStart(3)} (${String(dansGroupes).padStart(4)} écoles dedans, plus grosse ${String(plusGrosGroupe).padStart(3)}) | pages « coin » ${String(coins.length).padStart(3)} | écoles jamais vues seules, même de près ${String(invisibles).padStart(4)} | groupe le plus étendu ${Math.round(plusLarge)} m | calcul ${ms.toFixed(1)} ms`,
          );
        }
      }
    }
    (await import('node:fs')).writeFileSync(process.env.SIM_OUT as string, lignes.join('\n'));
  }, 600000);
});
